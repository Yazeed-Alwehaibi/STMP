-- Create Venue first
CREATE TABLE "Venue" (
  "venueID" SERIAL PRIMARY KEY,
  "venueName" VARCHAR(100),
  "repID" INT,
  "location" VARCHAR(100),
  "departments" VARCHAR(100)
);

-- Create Users after Venue
CREATE TABLE "Users" (
  "systemID" SERIAL PRIMARY KEY,
  "userID" INT UNIQUE,
  "userName" VARCHAR(255),
  "email" VARCHAR(50),
  "role" VARCHAR(30) CHECK ("role" IN ('Supervisor', 'Student', 'Training Representative')),
  "departmentOrMajor" VARCHAR(100),
  "extraInfo" TEXT,
  "status" VARCHAR(50),
  "password" VARCHAR(255),
  "venueID" INT
);

-- Add foreign key constraint separately after both tables exist
ALTER TABLE "Users" ADD CONSTRAINT "FK_user_venue" FOREIGN KEY ("venueID") REFERENCES "Venue"("venueID");
ALTER TABLE "Venue" ADD CONSTRAINT "FK_venue_rep" FOREIGN KEY ("repID") REFERENCES "Users"("systemID");

-- Create Applications table
CREATE TABLE "Applications" (
  "applicationID" SERIAL PRIMARY KEY,
  "supervisorID" INT,
  "studentID" INT,
  "repID" INT,
  "venueID" INT,
  "startDate" TIMESTAMP,
  "endDate" TIMESTAMP,
  "content_file" TEXT,
  "status" VARCHAR(50),
  "type" VARCHAR(50),
  CONSTRAINT "FK_application_rep" FOREIGN KEY ("repID") REFERENCES "Users"("systemID"),
  CONSTRAINT "FK_application_supervisor" FOREIGN KEY ("supervisorID") REFERENCES "Users"("systemID"),
  CONSTRAINT "FK_application_student" FOREIGN KEY ("studentID") REFERENCES "Users"("systemID"),
  CONSTRAINT "FK_application_venue" FOREIGN KEY ("venueID") REFERENCES "Venue"("venueID")
);

-- Create Reports table
CREATE TABLE "Reports" (
  "reportID" SERIAL PRIMARY KEY,
  "studentID" INT,
  "supervisorID" INT,
  "applicationID" INT,
  "submissionDate" TIMESTAMP,
  "mark" DECIMAL(4, 2),
  "feedback" TEXT,
  "type" VARCHAR(10) CHECK ("type" IN ('1', '2', '3', 'final')),
  "status" VARCHAR(250),
  "content" TEXT
);

-- Create Presentations table
CREATE TABLE "Presentations" (
  "presentationID" SERIAL PRIMARY KEY,
  "studentID" INT,
  "supervisorID" INT,
  "applicationID" INT,
  "date" TIMESTAMP,
  "file" TEXT,
  "mark" DECIMAL(4, 2),
  "status" VARCHAR(50)
);

-- Create Feedback table
CREATE TABLE "Feedback" (
  "feedbackID" SERIAL PRIMARY KEY,
  "studentID" INT,
  "venueID" INT,
  "applicationID" INT,
  "content" TEXT,
  "status" VARCHAR(50)
);

-- Create Offers table
CREATE TABLE "Offers" (
  "offerID" SERIAL PRIMARY KEY,
  "venueID" INT,
  "repID" INT,
  "title" VARCHAR(100),
  "description" TEXT,
  "startDate" TIMESTAMP,
  "endDate" TIMESTAMP,
  "maxParticipant" INT,
  "status" VARCHAR(50),
  CONSTRAINT "FK_offers_rep" FOREIGN KEY ("repID") REFERENCES "Users"("systemID"),
  CONSTRAINT "FK_offers_venue" FOREIGN KEY ("venueID") REFERENCES "Venue"("venueID")
);

-- Create Community table
CREATE TABLE "Community" (
  "communityID" SERIAL PRIMARY KEY,
  "communityName" VARCHAR(100) UNIQUE
);

-- Create Members table
CREATE TABLE "Members" (
  "communityID" INT,
  "systemID" INT,
  PRIMARY KEY ("communityID", "systemID"),
  CONSTRAINT "FK_members_community" FOREIGN KEY ("communityID") REFERENCES "Community"("communityID"),
  CONSTRAINT "FK_members_system" FOREIGN KEY ("systemID") REFERENCES "Users"("systemID")
);

-- Create Content table
CREATE TABLE "Content" (
  "contentID" SERIAL PRIMARY KEY,
  "communityID" INT,
  "systemID" INT,
  "content" TEXT,
  "date_time" TIMESTAMP,
  CONSTRAINT "FK_content_community" FOREIGN KEY ("communityID") REFERENCES "Community"("communityID"),
  CONSTRAINT "FK_content_system" FOREIGN KEY ("systemID") REFERENCES "Users"("systemID")
);
